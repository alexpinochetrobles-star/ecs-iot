<script>
function getCookie(name) {
  return document.cookie.split('; ').find(r => r.startsWith(name + '='))?.split('=')[1];
}

const ctx = document.getElementById('graficoPozo').getContext('2d');
const grafico = new Chart(ctx, {
  type: 'line',
  data: {
    labels: [],
    datasets: [
      { label: 'Nivel Estático (m)',  data: [], borderColor: '#0d6efd', tension: .3 },
      { label: 'Nivel Dinámico (m)',  data: [], borderColor: '#ffc107', tension: .3 },
      { label: 'Caudal (L/s)',        data: [], borderColor: '#198754', yAxisID: 'y2', tension: .3 },
    ],
  },
  options: {
    responsive: true, maintainAspectRatio: false,
    scales: {
      y:  { title: { display: true, text: 'Nivel (m)' }, reverse: false },
      y2: { position: 'right', title: { display: true, text: 'Caudal (L/s)' } },
    },
  },
});

async function actualizarDashboard() {
  const r1 = await fetch('/api/lecturas/');
  const d1 = await r1.json();
  grafico.data.labels = d1.labels;
  grafico.data.datasets[0].data = d1.nivel_estatico;
  grafico.data.datasets[1].data = d1.nivel_dinamico;
  grafico.data.datasets[2].data = d1.caudal;
  grafico.update('none');

  document.getElementById('cardNE').textContent = (d1.ultima.nivel_estatico_m ?? '--') + ' m';
  document.getElementById('cardND').textContent = (d1.ultima.nivel_dinamico_m ?? '--') + ' m';
  document.getElementById('cardQ').textContent  = (d1.ultima.caudal_lps ?? '--') + ' L/s';
  document.getElementById('horaUltima').textContent = d1.ultima.fecha_hora || '--';

  const r2 = await fetch('/api/estado-gpio/');
  const d2 = await r2.json();

  const bomba = document.getElementById('cardBomba');
  bomba.textContent = d2.bomba_encendida ? 'ON' : 'OFF';
  bomba.className = 'mb-0 ' + (d2.bomba_encendida ? 'text-success' : 'text-danger');

  const bN = document.getElementById('badgeNivelMin');
  bN.textContent = d2.nivel_minimo_ok ? 'Nivel OK' : 'Nivel BAJO';
  bN.className = 'badge me-2 ' + (d2.nivel_minimo_ok ? 'bg-success' : 'bg-danger');

  const bP = document.getElementById('badgeParo');
  bP.textContent = d2.paro_emergencia_activo ? 'PARO ACTIVO' : 'Paro OK';
  bP.className = 'badge ' + (d2.paro_emergencia_activo ? 'bg-danger' : 'bg-success');
}

async function controlarBomba(accion) {
  const r = await fetch('/api/bomba/', {
    method: 'POST',
    headers: { 'X-CSRFToken': getCookie('csrftoken'),
               'Content-Type': 'application/x-www-form-urlencoded' },
    body: 'accion=' + accion,
  });
  const d = await r.json();
  document.getElementById('msgControl').textContent = d.mensaje;
}

document.getElementById('btnOn')?.addEventListener('click',  () => controlarBomba('on'));
document.getElementById('btnOff')?.addEventListener('click', () => controlarBomba('off'));

setInterval(actualizarDashboard, 3000);
actualizarDashboard();
</script>
